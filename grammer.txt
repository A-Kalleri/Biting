Bitter :
        1               :      1
        0               :      0

        |               :      or
        &               :      and
        ^               :      xor
        ~/!             :      not

        x<numbers>      :      variable, x indicates variable.

        ><variable>     :      store into variable.
        <<variable>     :      print variable.
        .               :      reset lhs

        rp/RP           :       print register
        
        operations accumulate into reg_lhs

start -> fill_buffer -> lexer -> parser -> end
              ^           |
              |___________|


program         := statement*

statement       := expression
                 | expression > identifier
                 | reserved_register > identifier
                 | <

expression      := operand
                 | operand (operator operand)*
                 | reserved_register(lhs) NOT

operator        := |
                 | &
                 | ^

operand         := 0
                 | 1
                 | identifer

future:-
        #<variable>     can store a single ascii chars and only be printed uses the same symbol table as of x variables
        <0 <variable>   prints if lhs is 0/false
        <1 <variable>   prints if lhs is 1/true
