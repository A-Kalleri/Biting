program         := statement*

statement       := expression
                 | expression > identifier
                 | reserved_register > identifier

expression      := operand
                 | operand operator operand

operand         := 0
                 | 1
                 | identifer
                 | reserved_register(lhs) NOT

operator        := |
                 | &
                 | ^
